# ZimZilla — the harness's dependencies.
#
#   python -m venv .venv && source .venv/bin/activate
#   pip install -r requirements.txt
#
# This installs the harness's dependencies and zimzilla itself (the trailing
# `-e .`), which drops a `zimzilla` command into the venv's bin.
#
# The LiteLLM proxy is deliberately NOT in this file. It cannot share this
# environment: litellm[proxy] requires rich<14.0, while textual 8.x requires
# rich>=14.2. A requirements.txt that lists both is unsatisfiable — pip reports
# ResolutionImpossible and installs *nothing*, which is why
# `pip install -r requirements.txt` failed on a fresh checkout. The proxy gets
# its own venv instead (~/.zimzilla/venv, built by setup.sh), which
# start-litellm.sh already looks in.

anthropic>=0.40
textual>=0.80
# textual 8.x needs rich>=14.2. This floor holds only because litellm[proxy] —
# which pins rich<14 and would downgrade it — is kept out of this environment.
rich>=14.2.0
pyyaml>=6.0

# zimzilla itself, editable, so the `zimzilla` console script is installed and
# the package is importable from any directory.
-e .
